cmake_minimum_required(VERSION 3.24)
project(TurnHubLVGL VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
  # LVGL 9.3's MSVC global-data exports require a shared library.
  set(BUILD_SHARED_LIBS ON CACHE BOOL "Shared LVGL on MSVC" FORCE)
endif()
option(TURNHUB_SANITIZERS "Instrument the desktop build with ASan/UBSan" OFF)
if(TURNHUB_SANITIZERS)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  else()
    message(FATAL_ERROR "TURNHUB_SANITIZERS needs GCC or Clang")
  endif()
endif()
include(FetchContent)
option(TURNHUB_SYSTEM_SDL "Use an installed exact SDL 2.32.4 instead of building it" OFF)
if(TURNHUB_SYSTEM_SDL)
  find_package(SDL2 2.32.4 EXACT CONFIG REQUIRED)
else()
  if(MSVC)
    # Shared LVGL would embed a private static SDL, leaving the executable with a
    # second, uninitialized copy (no events, renderer or watches); share one DLL.
    set(SDL_SHARED ON CACHE BOOL "" FORCE)
    set(SDL_STATIC OFF CACHE BOOL "" FORCE)
  else()
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
  endif()
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  # Software renderer is sufficient; no GPU SDK is required.
  set(SDL_VULKAN OFF CACHE BOOL "" FORCE)
  set(SDL_OPENGL OFF CACHE BOOL "" FORCE)
  set(SDL_OPENGLES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL2
    URL https://github.com/libsdl-org/SDL/releases/download/release-2.32.4/SDL2-2.32.4.tar.gz
    URL_HASH SHA256=f15b478253e1ff6dac62257ded225ff4e7d0c5230204ac3450f1144ee806f934
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(SDL2)
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(lvgl
  # v9.3.0, immutable release commit.
  URL https://codeload.github.com/lvgl/lvgl/tar.gz/c033a98afddd65aaafeebea625382a94020fe4a7
  URL_HASH SHA256=9c6f8230d0b0e11141aad8f047af699322dbd3d37b2c41170b1526c12d5dfbb3
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(lvgl)
if(TARGET SDL2::SDL2-static AND NOT TURNHUB_SYSTEM_SDL)
  set(TH_SDL SDL2::SDL2-static)
else()
  set(TH_SDL SDL2::SDL2)
endif()
target_link_libraries(lvgl PUBLIC ${TH_SDL})
if(TURNHUB_SYSTEM_SDL)
  target_include_directories(lvgl SYSTEM PUBLIC ${SDL2_INCLUDE_DIRS})
endif()
file(GLOB TH_FONTS CONFIGURE_DEPENDS ../shared/display/th_*.c)
add_library(turnhub_ui STATIC src/ui.cpp src/assets.cpp ${TH_FONTS})
target_include_directories(turnhub_ui PUBLIC include ../shared/include PRIVATE ../Atlas/include)
target_link_libraries(turnhub_ui PUBLIC lvgl::lvgl)
add_executable(turnhub_sim src/main.cpp src/sdl_input.cpp src/fixtures.cpp)
target_link_libraries(turnhub_sim PRIVATE turnhub_ui)
if(MSVC)
  target_compile_options(turnhub_ui PRIVATE /W4)
  target_compile_options(turnhub_sim PRIVATE /W4)
else()
  target_compile_options(turnhub_ui PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(turnhub_sim PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME atlas_ui COMMAND turnhub_sim --device atlas --self-test)
  add_test(NAME sigil_ui COMMAND turnhub_sim --device sigil --self-test)
  set_tests_properties(atlas_ui sigil_ui PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy" TIMEOUT 20)
endif()
